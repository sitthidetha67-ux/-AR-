    function roundRect(ctx, x, y, w, h, r){
      ctx.beginPath();
      ctx.moveTo(x+r, y);
      ctx.arcTo(x+w, y, x+w, y+h, r);
      ctx.arcTo(x+w, y+h, x, y+h, r);
      ctx.arcTo(x, y+h, x, y, r);
      ctx.arcTo(x, y, x+w, y, r);
      ctx.closePath();
    }

    function chooseRoundItems(){
       ใน 1 ข้อ คำกริยา 1 คำ + คำนามหลอก 2 คำ
      const verbs = WORD_BANK.filter(w = w.type === กริยา);
      const nouns = WORD_BANK.filter(w = w.type === คำนาม);

      const verb = pickRandom(verbs);
      const nounPool = shuffleArray([...nouns]);
      const distractors = nounPool.slice(0, 2);

      return shuffleArray([verb, ...distractors]);
    }

    function pickRandom(arr){
      return arr[Math.floor(Math.random()  arr.length)];
    }

    function shuffleArray(arr){
      for(let i = arr.length - 1; i  0; i--){
        const j = Math.floor(Math.random()  (i + 1));
        [arr[i], arr[j]] = [arr[j], arr[i]];
      }
      return arr;
    }

    }

    function burst(x, y, color){
      for(let i=0;i18;i++){
        const mat = new THREE.MeshBasicMaterial({ color, transparenttrue, opacity1 });
        const mesh = new THREE.Mesh(new THREE.SphereGeometry(4 + Math.random()4, 8, 8), mat);
        mesh.position.set(x, y, 40);
        scene.add(mesh);
        const angle = Math.random()  Math.PI  2;
        const speed = 90 + Math.random()230;
        particles.push({
          mesh,
          vel new THREE.Vector2(Math.cos(angle)speed, Math.sin(angle)speed),
          life .65 + Math.random().35
        });
      }
    }

    function updateParticles(dt){
      for(let i=particles.length-1;i=0;i--){
        const p = particles[i];
        p.life -= dt;
        p.mesh.position.x += p.vel.x  dt;
        p.mesh.position.y += p.vel.y  dt;
        p.vel.y += 200  dt;
        p.mesh.material.opacity = Math.max(0, p.life);
        if(p.life = 0){
          scene.remove(p.mesh);
          particles.splice(i,1);
        }
      }
    }

    function createProjectile(origin, dir, strength){
      const mesh = createArrowMesh();
      mesh.position.set(origin.x + dir.x  50, origin.y + dir.y  50, 50);
      mesh.rotation.z = Math.atan2(dir.y, dir.x);
      scene.add(mesh);

      const speed = ARROW_BASE_SPEED + ARROW_EXTRA_SPEED  strength;
      arrows.push({
        mesh,
        pos new THREE.Vector2(mesh.position.x, mesh.position.y),
        vel dir.clone().multiplyScalar(speed),
        life 3.8,
        strength
      });
    }

    function updateArrows(dt){
      for(let i=arrows.length-1;i=0;i--){
        const a = arrows[i];
        a.life -= dt;
        a.pos.addScaledVector(a.vel, dt);
        a.mesh.position.x = a.pos.x;
        a.mesh.position.y = a.pos.y;
        a.mesh.rotation.z = Math.atan2(a.vel.y, a.vel.x);

        let hitFruit = null;
        for(const fruit of activeFruits){
          const d = a.pos.distanceTo(fruit.pos);
          if(d  fruit.radius){
            hitFruit = fruit;
            break;
          }
        }

        if(hitFruit){
          scene.remove(a.mesh);
          arrows.splice(i,1);
          resolveFruit(hitFruit, true);
          continue;
        }

        if(a.life = 0  a.pos.x  -160  a.pos.x  W + 160  a.pos.y  -160  a.pos.y  H + 160){
          scene.remove(a.mesh);
          arrows.splice(i,1);
        }
      }
    }

    function averagePoint(arr){
      return {

    function distNorm(a,b){
      const dx = a.x - b.x;
      const dy = a.y - b.y;
      return Math.sqrt(dxdx + dydy);
    }

    function detectGesture(lm){
       ใช้วิธี heuristic นิ้วเหยียดเมื่อปลายนิ้วอยู่ไกลจากข้อมือมากกว่าข้อกลาง
      const wrist = lm[0];
      const fingers = [
        { tip8, pip6 },
        { tip12, pip10 },
        { tip16, pip14 },
        { tip20, pip18 }
      ];

      let extended = 0;
      for(const f of fingers){
        if(distNorm(lm[f.tip], wrist)  distNorm(lm[f.pip], wrist)  1.12) extended++;
      }

       นิ้วโป้งช่วยแยกแบมือกำมือ
      const thumbOpen = distNorm(lm[4], lm[9])  distNorm(lm[3], lm[9])  1.03;
      if(thumbOpen) extended += .5;

      if(extended = 3.3) return open;
      if(extended = 1.2) return fist;
      return half;
    }

    function getTargetStrengthFromPull(pullProjection){

      const raw = THREE.MathUtils.clamp((pullProjection - PULL_DEAD_ZONE)  (MAX_PULL - PULL_DEAD_ZONE), 0, 1);

    function smoothChargeToTarget(targetStrength, dt){
       ไม่ให้แรงยิงกระโดดตามมือทันที แต่ค่อย ๆ เพิ่มลดตามระยะดึง
      const speed = targetStrength  drawStrength  CHARGE_RISE_SPEED  CHARGE_FALL_SPEED;
      const maxStep = speed  dt;
      const diff = targetStrength - drawStrength;

      if(Math.abs(diff) = maxStep){
        drawStrength = targetStrength;
      }else{
        drawStrength += Math.sign(diff)  maxStep;
      }

      drawStrength = THREE.MathUtils.clamp(drawStrength, 0, 1);
    }
